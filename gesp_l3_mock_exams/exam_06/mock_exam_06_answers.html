编程题1参考代码

#include 
#include 
using namespace std;

int main() {
    string s;
    getline(cin, s);
    int upper=0, lower=0, digit=0, space=0, other=0;
    for (int i = 0; i < s.length(); i++) {
        if (s[i] >= 'A' && s[i] <= 'Z') upper++;
        else if (s[i] >= 'a' && s[i] <= 'z') lower++;
        else if (s[i] >= '0' && s[i] <= '9') digit++;
        else if (s[i] == ' ') space++;
        else other++;
    }
    cout << upper << endl << lower << endl << digit << endl << space << endl << other << endl;
    return 0;
}
