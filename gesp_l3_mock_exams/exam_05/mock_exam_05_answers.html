#include 
using namespace std;

int main() {
    int m, k, n;
    cin >> m >> k >> n;
    int a[100][100], b[100][100], c[100][100];
    for (int i = 0; i < m; i++)
        for (int j = 0; j < k; j++)
            cin >> a[i][j];
    for (int i = 0; i < k; i++)
        for (int j = 0; j < n; j++)
            cin >> b[i][j];
    for (int i = 0; i < m; i++)
        for (int j = 0; j < n; j++) {
            c[i][j] = 0;
            for (int p = 0; p < k; p++)
                c[i][j] += a[i][p] * b[p][j];
        }
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++)
            cout << c[i][j] << (j < n-1 ? " " : "");
        cout << endl;
    }
    return 0;
}
