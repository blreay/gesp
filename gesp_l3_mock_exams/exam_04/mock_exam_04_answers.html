编程题1参考代码

#include 
using namespace std;

int main() {
    long long n;
    cin >> n;
    cout << n << "=";
    bool first = true;
    for (long long i = 2; i * i <= n; i++) {
        while (n % i == 0) {
            if (!first) cout << "*";
            cout << i;
            first = false;
            n /= i;
        }
    }
    if (n > 1) {
        if (!first) cout << "*";
        cout << n;
    }
    cout << endl;
    return 0;
}

#include 
#include 
using namespace std;

int main() {
    int n;
    cin >> n;
    int a[20][20] = {};
    int dx[] = {0, 1, 0, -1};
    int dy[] = {1, 0, -1, 0};
    int x = 0, y = 0, dir = 0;
    for (int i = 1; i <= n * n; i++) {
        a[x][y] = i;
        int nx = x + dx[dir], ny = y + dy[dir];
        if (nx < 0 || nx >= n || ny < 0 || ny >= n || a[nx][ny] != 0) {
            dir = (dir + 1) % 4;
            nx = x + dx[dir];
            ny = y + dy[dir];
        }
        x = nx; y = ny;
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++)
            cout << setw(4) << a[i][j];
        cout << endl;
    }
    return 0;
}
